#include <ESP8266WiFi.h>
#include <PubSubClient.h>

const char* ssid = "NAMA_WIFI";
const char* password = "PASSWORD_WIFI";
const char* mqtt_server = "broker.hivemq.com";
const int mqtt_port = 1883;

const char* TOPIC_STATUS = "hasbielektro-ui-2026/esp8266/status";
const char* TOPIC_DATA   = "hasbielektro-ui-2026/esp8266/data";

WiFiClient espClient;
PubSubClient client(espClient);
unsigned long lastSend = 0;
int counter = 0;

void setupWiFi() {
  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password);
  pinMode(LED_BUILTIN, OUTPUT);
  while (WiFi.status() != WL_CONNECTED) {
    digitalWrite(LED_BUILTIN, LOW);
    delay(200);
    digitalWrite(LED_BUILTIN, HIGH);
    delay(200);
  }
  digitalWrite(LED_BUILTIN, LOW);
}

void reconnectMQTT() {
  while (!client.connected()) {
    String clientId = "ESP8266-" + String(random(0xffff), HEX);
    if (client.connect(clientId.c_str())) {
      client.publish(TOPIC_STATUS, "ONLINE (Charger HP)", true);
    } else {
      delay(2000);
    }
  }
}

void setup() {
  setupWiFi();
  client.setServer(mqtt_server, mqtt_port);
}

void loop() {
  if (WiFi.status() != WL_CONNECTED) setupWiFi();
  if (!client.connected()) reconnectMQTT();
  client.loop();

  if (millis() - lastSend >= 5000) {
    lastSend = millis();
    counter++;
    String payload = "Aktif #" + String(counter) + " | RSSI: " + String(WiFi.RSSI()) + " dBm";
    client.publish(TOPIC_STATUS, "ONLINE (Charger HP)", true);
    client.publish(TOPIC_DATA, payload.c_str(), true);
  }
}
